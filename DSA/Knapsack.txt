
import java.util.Arrays;

public class KnapsackDP {

    // Recursive Approach (0/1 Knapsack)
    public static int maxStealZeroOne(int W, int j, int[] weights, double[] values) {
        assert j >= 0;
        assert weights.length == values.length;

        // Base cases
        if (W == 0) return 0; // No weight left
        if (W < 0) return Integer.MIN_VALUE; // Exceeded weight limit
        if (j >= weights.length) return 0; // No more items left

        // Either take item j or skip it
        return (int) Math.max(
            values[j] + maxStealZeroOne(W - weights[j], j + 1, weights, values),  // Take item j
            maxStealZeroOne(W, j + 1, weights, values)  // Skip item j
        );
    }

    // Memoized Dynamic Programming Approach
    public static int memoizedMaxStealZeroOne(int W, int[] weights, double[] values) {
        int n = weights.length;
        assert values.length == n : "Weights and Values list must be of same size";
        assert W >= 0;

        if (W == 0) return 0; // No weight to steal

        // DP table to store max value for each (weight, item) pair
        double[][] T = new double[W + 1][n];
        int[][] S = new int[W + 1][n]; // Stores choices (1 = taken, 0 = skipped)

        // Fill DP table
        for (int w = 1; w <= W; w++) {
            for (int j = n - 1; j >= 0; j--) {
                double take = (w >= weights[j]) ? values[j] + getTableEntry(T, w - weights[j], j + 1) : Integer.MIN_VALUE;
                double skip = getTableEntry(T, w, j + 1);

                if (take > skip) {
                    T[w][j] = take;
                    S[w][j] = 1; // Item taken
                } else {
                    T[w][j] = skip;
                    S[w][j] = 0; // Item skipped
                }
            }
        }

        // Recover solution (items to steal)
        int weightLeft = W;
        System.out.println("Items Stolen:");
        for (int j = 0; j < n; j++) {
            if (S[weightLeft][j] == 1) {
                System.out.println("Item " + j + ": Weight = " + weights[j] + ", Value = " + values[j]);
                weightLeft -= weights[j];
            }
        }
        System.out.println("Total Weight Stolen: " + (W - weightLeft) + ", Value = " + T[W][0]);

        return (int) T[W][0];
    }

    // Helper function to safely get table entry (avoiding out-of-bounds)
    private static double getTableEntry(double[][] T, int w, int j) {
        if (w < 0) return Integer.MIN_VALUE;
        if (j >= T[0].length) return 0;
        return T[w][j];
    }

    // Main function to test the program
    public static void main(String[] args) {
        int W = 200;
        int[] weights = {1, 5, 20, 35, 90};
        double[] values = {15, 14.5, 19.2, 19.8, 195.2};

        // Test recursive approach
        System.out.println("Maximum Value (Recursive): " + maxStealZeroOne(W, 0, weights, values));

        // Test memoized DP approach
        System.out.println("Maximum Value (Memoized DP): " + memoizedMaxStealZeroOne(W, weights, values));
    }
}

Explanation
	1.	Recursive Approach (maxStealZeroOne):
	•	Tries both taking and skipping each item.
	•	Returns the maximum value possible.
	•	Has exponential time complexity O(2^n).
	2.	Memoized DP Approach (memoizedMaxStealZeroOne):
	•	Uses a table T[w][j] to store results.
	•	Uses a table S[w][j] to track which items were taken.
	•	Computes results iteratively.
	•	Recovers the list of items selected.
	•	Has polynomial time complexity O(nW).
